```c
/* v29.68: ReBAR 8 GiB unlock — XVE family, plain MMIO, same domain as the
 * verified Gen2 overrides (no Falcon privilege involved). Ported from
 * 40HX-Unlock-Drive-Patch 0003-cmp40hx-rebar-unlock (reference copy:
 * reference/40HX-Unlock-Drive-Patch/patches/615.78.08/) plus the Windows EFI
 * (unlock40x_v70.c [11]) which adds the standard PCIe Resizable BAR control:
 *   XVE_CYA 0x88724 = 0x30          override unlock
 *   XVE_CFG 0x88dcc  selector 7 + enable(bit31) -> offered cap 0x400->0x7fc00
 *   RBAR-CTL 0x88bc0 (cfg 0xBC0 mirror) bits[13:8] = log2(MiB), 13 = 8 GiB
 * v70 notes: skipping RBAR-CTL leaves BAR1 at 64 MiB. Must run AFTER the last
 * SBR (SBR clears endpoint PCI state). Readback-verified; any mismatch on a
 * written field reverts the whole set, so the failure mode is only
 * "ReBAR stays off". */
#define XVE_CYA_OFF         0x0088724U
#define XVE_CAP_OFF         0x0088BBCU
#define XVE_RBAR_CTL_OFF    0x0088BC0U
#define XVE_CFG_OFF         0x0088DCCU
#define XVE_CYA_UNLOCK      0x00000030U
#define XVE_CFG_SEL_MASK    0x0000000FU
#define XVE_CFG_ENABLE      0x80000000U
#define XVE_CFG_SELECTOR    7U           /* selector 7 = 8 GiB (2^(6+7) MiB) */
#define XVE_RBAR_SIZE_MASK  0x00003F00U  /* bits [13:8] */
#define XVE_RBAR_SIZE_SHIFT 8U
#define REBAR_LOG2_MIB      13U          /* 2^13 MiB = 8 GiB */
#define REBAR_CAP_8GIB_BIT  (1U << 13)

const char nvperm_core_version[] = "40hx core v29.68 (compute+gen2+rebar)";

static int rebar_unlock(struct gpu_dev *g)
{
    uint32_t cya, cfg, ctl;
    uint32_t cap_before, cap_after;
    int fail = 0;

    cya = RD(g, XVE_CYA_OFF);
    cfg = RD(g, XVE_CFG_OFF);
    ctl = RD(g, XVE_RBAR_CTL_OFF);
    cap_before = RD(g, XVE_CAP_OFF);
    g->log(g, "rebar before: CYA=%08x CFG=%08x CAP=%08x CTL=%08x\n",
           cya, cfg, cap_before, ctl);

    WR(g, XVE_CYA_OFF, XVE_CYA_UNLOCK);
    WR(g, XVE_CFG_OFF,
       (cfg & ~XVE_CFG_SEL_MASK) | XVE_CFG_ENABLE | XVE_CFG_SELECTOR);
    WR(g, XVE_RBAR_CTL_OFF,
       (ctl & ~XVE_RBAR_SIZE_MASK) | (REBAR_LOG2_MIB << XVE_RBAR_SIZE_SHIFT));

    /* Only our own written fields gate success; the capability readback is
     * informational (some cards may update it asynchronously). */
    if (RD(g, XVE_CYA_OFF) != XVE_CYA_UNLOCK)
        fail = 1;
    if ((RD(g, XVE_CFG_OFF) & (XVE_CFG_ENABLE | XVE_CFG_SEL_MASK)) !=
        (XVE_CFG_ENABLE | XVE_CFG_SELECTOR))
        fail = 1;
    if ((RD(g, XVE_RBAR_CTL_OFF) & XVE_RBAR_SIZE_MASK) !=
        (REBAR_LOG2_MIB << XVE_RBAR_SIZE_SHIFT))
        fail = 1;
    cap_after = RD(g, XVE_CAP_OFF);
    g->log(g, "rebar after:  CYA=%08x CFG=%08x CAP=%08x CTL=%08x%s\n",
           RD(g, XVE_CYA_OFF), RD(g, XVE_CFG_OFF), cap_after,
           RD(g, XVE_RBAR_CTL_OFF), fail ? " (MISMATCH)" : "");
    if (!(cap_after & REBAR_CAP_8GIB_BIT))
        g->log(g, "rebar note: CAP bit13 (8 GiB offered) not visible yet; "
                  "check lspci Resizable-BAR sizes after boot\n");
    if (fail) {
        WR(g, XVE_CFG_OFF, cfg);
        WR(g, XVE_RBAR_CTL_OFF, ctl);
        WR(g, XVE_CYA_OFF, cya);
        g->log(g, "rebar: readback mismatch, reverted to stock "
                  "(BAR1 stays 64 MiB; harmless)\n");
        return -1;
    }
    g->log(g, "rebar: XVE aperture unlocked to 8 GiB; OS resize follows\n");
    return 0;
}
```

### 4.3 "原理速览"新增 ReBAR 节（放在 Gen2 节之后）

### ReBAR（8 GiB BAR1 aperture，v29.68 新增）

三个普通 MMIO 写（与 Gen2 同属 XVE 族，**无需 Booter 特权**），必须在最后一次 SBR 之后执行：

| BAR0 偏移 | 写入 | 作用 |
|---|---|---|
| `0x88724` CYA | `0x30` | XVE override 解锁 |
| `0x88DCC` CFG | selector 7 + bit31 | 可选尺寸表 `0x400` → `0x7fc00`（8 GiB 进入列表） |
| `0x88BC0` RBAR-CTL | bits[13:8]=13 | 标准 PCIe Resizable BAR 控制寄存器，选定 `2^13 MiB = 8 GiB` |

全部回读校验；任何不一致回写原值并报失败（失败形态 = ReBAR 没开，无副作用）。
Linux 侧接手：`NVreg_EnableResizableBar=1`（原生驱动模块参数，非改驱动）→
`nv_resize_pcie_bars` → `pci_resize_resource` 按标准 RBAR 能力完成 BAR1 重分配。
寄存器值移植自驱动补丁 0003（参考拷贝在 `reference/40HX-Unlock-Drive-Patch/patches/615.78.08/`）。
注意：**"UEFI 解锁 + 原生驱动 resize"这个组合未经上游验证**（上游 Linux 走的是内核补丁），
以本机实测为准；若失败最坏结果是 BAR1 维持 64 MiB。

> 算力（SS0/SS1）与 ReBAR 是两回事：选择子写入实测被 SEC2 权限层丢弃，必须走 Booter 特权
> （路线 B 文档 §1），ReBAR 这套 XVE 写则是普通 MMIO 就能粘住的。

## 5. 工具存档：`tools/read_bar0.py`（完整源码备份）

> 用途：进 Linux、驱动加载后从 BAR0 读回解锁寄存器并自动判读——
> **B2 存活侦察（rmmod/modprobe 前后对比 SS0/SS1）的唯一工具**，回退时务必保留。

```python
#!/usr/bin/env python3
"""Read the CMP 40HX unlock registers out of BAR0 while Linux is running.

    sudo python3 tools/read_bar0.py [BDF]

The UEFI stage writes the compute selectors long before the NVIDIA driver
loads.  This tool answers the one question that matters: did those writes
survive GSP/SEC2 initialisation, or did the driver reset them back to the
factory values?

Registers are read strictly in the order listed and each line is flushed
immediately, so if the probe ever hangs on a protected region the last line
printed tells you exactly which offset was reached.
"""

import mmap
import os
import struct
import sys

VENDOR_DEVICE = (0x10DE, (0x1F0B, 0x220D))

# name, bar0 offset, value seen when locked, value the unlock wants
REGISTERS = [
    ("BOOT0",          0x000000, None,          None),
    ("SS0",            0x409664, 0x00000000,    0x88888888),
    ("SS0_RD",         0x409668, None,          None),
    ("SS1",            0x40966C, 0x00000000,    0x00000008),
    ("FECS_PLM",       0x409650, None,          0xFFFFFF8F),
    ("SEC2_RESET_PLM", 0x8403C4, 0x0000008F,    0x000000FF),
    ("XP3G_PLM",       0x8E1B0,  None,          0xFFFFFFFF),
    ("XVE_08610",      0x088610, None,          0x00001001),
    ("XVE_OVR",        0x08872C, None,          0x00000006),
    ("XVE_PCI_CAP",    0x088084, None,          None),
    ("XVE_PCI_LNK",    0x088088, None,          None),
    ("XVE_PCI_CAP2",   0x0880A4, 0x00000002,    0x00000006),
    ("XVE_PCI_CTL2",   0x0880A8, None,          None),
    ("GEN2_CFG",       0x08C040, None,          None),
    ("GEN2_PL",        0x08C1C0, None,          None),
    ("GEN2_CYA",       0x08C2C0, None,          None),
    ("GEN2_MISC1",     0x08841C, None,          None),
    ("WPR_ADDR_LO",    0x1FA824, None,          None),
    ("WPR_ADDR_HI",    0x1FA828, None,          None),
]


def find_gpu():
    """Return the BDF of the first CMP 40HX found through sysfs."""
    base = "/sys/bus/pci/devices"
    try:
        entries = sorted(os.listdir(base))
    except OSError as error:
        die("cannot read %s: %s" % (base, error))
    for bdf in entries:
        path = os.path.join(base, bdf)
        try:
            vendor = int(open(os.path.join(path, "vendor")).read(), 16)
            device = int(open(os.path.join(path, "device")).read(), 16)
        except (OSError, ValueError):
            continue
        if vendor == VENDOR_DEVICE[0] and device in VENDOR_DEVICE[1]:
            return bdf
    die("no NVIDIA 10de:1f0b / 10de:220d device found; pass the BDF explicitly")


def die(message):
    sys.stderr.write("error: %s\n" % message)
    sys.exit(1)


def main():
    bdf = sys.argv[1] if len(sys.argv) > 1 else None
    if bdf and not bdf.startswith("0000:"):
        bdf = "0000:" + bdf

    if not bdf:
        bdf = find_gpu()
    if not os.path.isdir("/sys/bus/pci/devices/%s" % bdf):
        die("device %s does not exist" % bdf)

    resource = "/sys/bus/pci/devices/%s/resource0" % bdf
    try:
        fd = os.open(resource, os.O_RDWR | os.O_SYNC)
    except OSError as error:
        if error.errno == 13:
            die("permission denied on %s - run this with sudo" % resource)
        die("cannot open %s: %s" % (resource, error))

    size = os.fstat(fd).st_size
    if size <= 0x40966C:
        die("BAR0 is only 0x%x bytes; FECS window not exposed" % size)

    window = mmap.mmap(fd, size, mmap.MAP_SHARED,
                       mmap.PROT_READ | mmap.PROT_WRITE)

    print("CMP 40HX BAR0 readout  (%s, BAR0 %d MiB)" % (bdf, size >> 20))
    print("  driver: %s" % (os.readlink(
        "/sys/bus/pci/devices/%s/driver" % bdf).rsplit("/", 1)[-1]
        if os.path.islink("/sys/bus/pci/devices/%s/driver" % bdf)
        else "none bound"))
    print()
    print("  %-16s %-10s %-12s %s" %
          ("register", "offset", "read", "at target?"))
    print("  " + "-" * 48)

    values = {}
    for name, offset, locked, want in REGISTERS:
        if offset + 4 > size:
            continue
        value = struct.unpack("<I", window[offset:offset + 4])[0]
        values[name] = value
        if locked is None and want is None:
            state = "-"
        elif want is not None:
            state = "YES" if value == want else "no"
        else:
            state = "YES" if value != locked else "no"
        print("  %-16s %-10s %-12s %s" %
              (name, "0x%06X" % offset, "0x%08X" % value, state))
        sys.stdout.flush()

    print()
    report(values)
    window.close()
    os.close(fd)


def report(values):
    ss0 = values.get("SS0")
    ss1 = values.get("SS1")
    reset_plm = values.get("SEC2_RESET_PLM")
    cap2 = values.get("XVE_PCI_CAP2")

    print("  ---- verdict ----")

    if ss0 == 0x88888888 and ss1 == 0x00000008:
        print("  SS0/SS1 selectors are STILL SET after the driver loaded.")
        print("  The UEFI write survived, so the unlock path is not blocked at")
        print("  the register level.  If Tensor/FP16 throughput is still ~1")
        print("  TFLOPS, the selectors alone are not sufficient - check")
        print("  SEC2_RESET_PLM (want 0xFF, factory 0x8F) above.")
    elif ss0 in (0x00000000, 0xFFFFFFFF) and ss1 in (0x00000000, 0xFFFFFFFF):
        print("  SS0=0x%08X SS1=0x%08X - the selectors are back at their" %
              (ss0, ss1))
        print("  factory state.  GSP/SEC2 initialisation overwrote whatever the")
        print("  UEFI stage wrote, so a pre-driver MMIO write cannot unlock")
        print("  compute on this card.  The driver-side route writes them from")
        print("  inside the SEC2 booter privilege window for exactly this")
        print("  reason.")
        if ss0 == 0xFFFFFFFF:
            print("  (0xFFFFFFFF readback also matches 'region gated' - treat as")
            print("   protected rather than as a real value.)")
    else:
        print("  SS0=0x%08X SS1=0x%08X - neither the locked nor the unlocked" %
              (ss0, ss1))
        print("  reference value.  Capture the UEFI-side 'selector probe' lines")
        print("  from the same boot and compare, to find where it drifted.")

    if reset_plm is not None:
        if reset_plm == 0x000000FF:
            print("  SEC2_RESET_PLM = 0xFF: already at the unlocked value.")
        elif reset_plm == 0x0000008F:
            print("  SEC2_RESET_PLM = 0x8F: still factory.  The driver-side")
            print("  unlock also raises this to 0xFF; the UEFI stage never")
            print("  writes it.  Worth trying as a one-register experiment.")
        else:
            print("  SEC2_RESET_PLM = 0x%08X: unexpected value." % reset_plm)

    if cap2 is not None:
        print("  Link speed vector 0x%08X: Gen2 %s (%s)" %
              (cap2,
               "advertised" if cap2 & 0x04 else "NOT advertised",
               "this mirrors the policy, it is not writable directly"))


if __name__ == "__main__":
    main()
```

## 6. 核心结论速览（即使所有文档都被回退删掉，凭本节即可重建认知）

1. **算力（SS0/SS1 速度选择子）**：
   - 直接 MMIO 写**被 SEC2 权限层丢弃**（实测：想写 `0x88888888`/`0x8`，交棒与进系统都停在
     factory `0x00110000`/`0x00000006`；PDISP 读出 `0xBADF1100` 区域封锁标记）——手搓框架的
     `write_full_set()` 对 SS0/SS1 是空操作，只有 Gen2 部分真正生效。
   - 必须走 **SEC2 Booter 特权上下文**（把 NVIDIA 签名的 booter ucode/FWSEC 装回 SEC2，
     让它执行选择子 payload `v67_payload.bin`，在特权域内写 SS0/SS1/FECS_PLM/WPR2）。
   - cpu-rm 内核模块可行但**必须 GSP OFF**（`NVreg_EnableGpuFirmware=0`）——被"GSP 不能关"约束堵死；
     驱动补丁路线被"不改驱动"约束堵死；gnu-efi 编的 EFI（v70 与最小 hello 同配方）在本板固件
     **首行 Print 前就挂死**（GRUB chainloader 与 efibootmgr 直启都挂），出货 `40HXUNLK.EFI`
     也是 gnu-efi 编的（段表含 `.dynamic/.dynsym`），同样不能用。
   - **算力唯一活路 = B2**：把 v70 `booter_load_tu102_direct` 的 DIRECT_SEC2 compute 主线
     （几百行寄存器编排，非 8900 行全量）搬进本手搓框架 + 嵌入
     `v67_payload.bin / booter_ucode_prod.bin / fwsec_40hx_prod.bin` 三个签名资产。
     框架原语已齐（AllocatePages<4G DMA、MMIO、SBR、嵌入 ELF loader）。
   - **B2 开工前提 = 存活侦察**：Linux GSP-on 下
     `sudo python3 tools/read_bar0.py` → `rmmod nvidia*` → `modprobe nvidia` → 再读，
     判驱动 probe 是否把 SS0/SS1 重设回 factory。不重锁 → 出 B2 逐行移植规格；
     重锁 → B2 白干，须摊牌放宽约束。
2. **ReBAR（8 GiB BAR1 aperture）**：三个普通 MMIO 写即可（§2.2 的三个寄存器），**无需 Booter**；
   Linux 侧 `NVreg_EnableResizableBar=1` 由原生驱动完成重分配。"UEFI 解锁 + 原生 resize"
   组合上游未验证，实测为准；失败无害（= 没开）。
3. **Gen2**：已完成并稳定（两轮 CCC 兜底重训是对的，命中哪轮随机）。
4. **项目边界**：`40HX-Unlock-Drive-Patch` 仓库只读；借鉴一律复制进 `reference/`；
   改动一次一步、每步实测后再进下一步；不擅自开始 B2。
